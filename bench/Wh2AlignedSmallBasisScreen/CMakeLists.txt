function(read_pinned OUT PATH HASH)
    file(SHA256 "${PATH}" ACTUAL)
    if(NOT ACTUAL STREQUAL HASH)
        message(FATAL_ERROR "Re-audit observer reuse after source changes: ${PATH}")
    endif()
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PATH}")
    file(READ "${PATH}" VALUE)
    set(${OUT} "${VALUE}" PARENT_SCOPE)
endfunction()
function(replace_one VAR OLD NEW)
    string(FIND "${${VAR}}" "${OLD}" POS)
    if(POS LESS 0)
        message(FATAL_ERROR "Observer replacement missing: ${OLD}")
    endif()
    string(LENGTH "${OLD}" LENGTH)
    math(EXPR NEXT "${POS}+${LENGTH}")
    string(SUBSTRING "${${VAR}}" ${NEXT} -1 TAIL)
    string(FIND "${TAIL}" "${OLD}" AGAIN)
    if(NOT AGAIN LESS 0)
        message(FATAL_ERROR "Observer replacement not unique: ${OLD}")
    endif()
    string(REPLACE "${OLD}" "${NEW}" VALUE "${${VAR}}")
    set(${VAR} "${VALUE}" PARENT_SCOPE)
endfunction()
