cmake_minimum_required(VERSION 3.20)
project(Wh2SmallDormantCoreScreen LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
get_filename_component(BINARY "${CMAKE_CURRENT_BINARY_DIR}" REALPATH)
string(FIND "${BINARY}/" "${ROOT}/" INSIDE)
if(INSIDE EQUAL 0 OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use a fresh external 64-bit Linux observer build")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(SANITIZE "Instrument observer; use matching instrumented DSOs" OFF)
set(DSO_DIR "" CACHE PATH "Existing neutral-qualified baseline/candidate libraries")
if(NOT EXISTS "${DSO_DIR}/libbaseline.so" OR NOT EXISTS "${DSO_DIR}/libcandidate.so")
    message(FATAL_ERROR "Pass DSO_DIR: this project never builds codec libraries")
endif()
add_compile_options(-O3 -Wall -Wextra -Wpedantic -Werror -fno-lto)
add_link_options(-fno-lto)
if(SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
include_directories("${ROOT}/include")
set(SHARED "${ROOT}/bench/Wh2SmallPayload2/Screen.cpp")
file(SHA256 "${SHARED}" HASH)
if(NOT HASH STREQUAL "59a3e1d69fc2230bf1119ec0b8c209ef7f669ad1a613c96ee40a12f23b433494")
    message(FATAL_ERROR "Re-audit pinned fixture/work code before reuse")
endif()
file(READ "${SHARED}" WORKER)
string(FIND "${WORKER}" "struct Row {" CUT)
if(CUT LESS 0)
    message(FATAL_ERROR "Missing exact reused-code boundary")
endif()
string(SUBSTRING "${WORKER}" 0 ${CUT} WORKER)
function(replace_one OLD NEW)
    string(FIND "${WORKER}" "${OLD}" POS)
    if(POS LESS 0)
        message(FATAL_ERROR "Missing observer replacement: ${OLD}")
    endif()
    string(LENGTH "${OLD}" LENGTH)
    math(EXPR NEXT "${POS}+${LENGTH}")
    string(SUBSTRING "${WORKER}" ${NEXT} -1 TAIL)
    string(FIND "${TAIL}" "${OLD}" AGAIN)
    if(NOT AGAIN LESS 0)
        message(FATAL_ERROR "Non-unique observer replacement: ${OLD}")
    endif()
    string(REPLACE "${OLD}" "${NEW}" VALUE "${WORKER}")
    set(WORKER "${VALUE}" PARENT_SCOPE)
endfunction()
replace_one("Batch = 32, Reps = 12" "Batch = 64, Reps = 12")
replace_one("decltype(&wirehair_v2_encode) encode;" "decltype(&wirehair_v2_encoder_create_with_options) ordinary;\n    decltype(&wirehair_v2_profile_deserialize) parse;\n    decltype(&wirehair_v2_encode) encode;")
replace_one("LOAD(create, \"wirehair_v2_encoder_create_profile_id_with_options\");" "LOAD(create, \"wirehair_v2_encoder_create_profile_id_with_options\");\n        LOAD(ordinary, \"wirehair_v2_encoder_create_with_options\");\n        LOAD(parse, \"wirehair_v2_profile_deserialize\");")
replace_one("unsigned k, b, tail, policy;" "unsigned k, b, tail, policy, route;")
replace_one("const int r = lib->create(ProfileId(s.k), source, s.Message(), s.b, &opts, p.data(), 32, &n, &c);" "const uint64_t id = s.route >= 4 ? WIREHAIR_V2_PROFILE_CERTIFIED_2026_07 : ProfileId(s.k);\n        const int r = s.route == 0 ? lib->ordinary(source, s.Message(), s.b, &opts, p.data(), 32, &n, &c) :\n            lib->create(id, source, s.Message(), s.b, &opts, p.data(), 32, &n, &c);")
replace_one("void* prebuilt, Byte* output)" "void* prebuilt, Byte* output, Profile& p)")
replace_one("Handle h(api); Profile p = {};" "Handle h(api); p.fill(0);")
# Descriptor output belongs to the caller and is recorded, not guessed from
# the address of the expected/input descriptor. Zeroing stays inside WORK.
file(READ Main.inc MAIN)
file(WRITE "${BINARY}/Worker.cpp" "${WORKER}\n${MAIN}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${SHARED}" "${CMAKE_CURRENT_SOURCE_DIR}/Main.inc")
add_executable(worker "${BINARY}/Worker.cpp")
target_link_libraries(worker PRIVATE ${CMAKE_DL_LIBS})
enable_testing()
foreach(MODE neutral neutral-reverse)
    add_test(NAME ${MODE} COMMAND worker "${DSO_DIR}/libbaseline.so" "${DSO_DIR}/libcandidate.so" ${MODE})
    set_tests_properties(${MODE} PROPERTIES TIMEOUT 240)
    if(SANITIZE)
        set_tests_properties(${MODE} PROPERTIES ENVIRONMENT
            "ASAN_OPTIONS=detect_leaks=1:detect_stack_use_after_return=1;UBSAN_OPTIONS=halt_on_error=1")
    endif()
endforeach()
