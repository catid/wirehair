cmake_minimum_required(VERSION 3.20)
project(Wh2SmallDecoderStorage LANGUAGES C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
get_filename_component(BINARY "${CMAKE_CURRENT_BINARY_DIR}" REALPATH)
string(FIND "${BINARY}/" "${ROOT}/" INSIDE)
if(INSIDE EQUAL 0 OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use a fresh external 64-bit Linux qualification build")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PORTABLE "Force portable arithmetic (not a real non-GFNI host)" OFF)
option(SANITIZE "Instrument all codec and test objects" OFF)
add_compile_options(-O3 -Wall -Wextra -Wpedantic -Werror -fno-lto)
add_link_options(-fno-lto)
if(PORTABLE)
    add_compile_definitions(ANDROID=1)
endif()
if(SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
include_directories("${ROOT}/include" "${ROOT}/codec" "${ROOT}" "${ROOT}/test")
execute_process(COMMAND "${Python3_EXECUTABLE}" -B "${CMAKE_CURRENT_SOURCE_DIR}/Generate.py"
    "${ROOT}" "${BINARY}/candidate" RESULT_VARIABLE RESULT)
if(NOT RESULT EQUAL 0)
    message(FATAL_ERROR "Candidate generation failed")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/Generate.py" "${ROOT}/codec/WirehairSmallCore.h"
    "${ROOT}/test/V2SmallCodecTest.cpp" "${ROOT}/test/SmallCodecTest.cpp")
foreach(NAME WirehairSmallLookup.h WirehairSmallFacade.h)
    configure_file("${ROOT}/codec/${NAME}" "${BINARY}/candidate/${NAME}" COPYONLY)
endforeach()
# Retain the established screening-library link order: lookup/facade TUs,
# followed by V2Profile. Reordering unchanged TUs would add a layout change.
set(SMALL WirehairSmall.cpp WirehairSmallK5.cpp WirehairSmallK8.cpp WirehairV2Profile.cpp)
foreach(NAME IN LISTS SMALL)
    configure_file("${ROOT}/codec/${NAME}" "${BINARY}/candidate/${NAME}" COPYONLY)
    list(APPEND BASELINE_SOURCES "${ROOT}/codec/${NAME}")
    list(APPEND CANDIDATE_SOURCES "${BINARY}/candidate/${NAME}")
endforeach()
set(COMMON wirehair.cpp gf256.cpp WirehairCodec.cpp WirehairTools.cpp
    codec/WirehairK6.cpp codec/WirehairK6Core.cpp codec/WirehairV2Codec.cpp
    codec/WirehairV2Peel.cpp codec/WirehairV2Plan.cpp codec/WirehairV2Policy.cpp
    codec/WirehairV2Precode.cpp codec/WirehairV2PrecodeDecode.cpp
    codec/WirehairV2PrecodeEncode.cpp codec/WirehairV2Seeds.cpp codec/WirehairV2Solve.cpp)
list(TRANSFORM COMMON PREPEND "${ROOT}/")
add_library(common OBJECT ${COMMON})
add_library(baseline_objects OBJECT ${BASELINE_SOURCES})
add_library(candidate_objects OBJECT ${CANDIDATE_SOURCES})
foreach(TARGET common baseline_objects candidate_objects)
    set_target_properties(${TARGET} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_definitions(${TARGET} PRIVATE WIREHAIR_BUILDING=1)
endforeach()
enable_testing()
foreach(ARM baseline candidate)
    add_library(${ARM}_static STATIC $<TARGET_OBJECTS:common> $<TARGET_OBJECTS:${ARM}_objects>)
    target_link_libraries(${ARM}_static PUBLIC Threads::Threads)
    add_library(${ARM} SHARED $<TARGET_OBJECTS:common> $<TARGET_OBJECTS:${ARM}_objects>)
    target_link_libraries(${ARM} PRIVATE Threads::Threads)
    target_link_options(${ARM} PRIVATE "-Wl,--version-script=${ROOT}/abi/wirehair.map")
    if(ARM STREQUAL "baseline")
        set(TEST_SOURCE "${ROOT}/test/V2SmallCodecTest.cpp")
        set(SMALL_TEST_SOURCE "${ROOT}/test/SmallCodecTest.cpp")
        set(PROFILE_SOURCE "${ROOT}/codec/WirehairV2Profile.cpp")
    else()
        set(TEST_SOURCE "${BINARY}/candidate/V2SmallCodecTest.cpp")
        set(SMALL_TEST_SOURCE "${BINARY}/candidate/SmallCodecTest.cpp")
        set(PROFILE_SOURCE "${BINARY}/candidate/WirehairV2Profile.cpp")
    endif()
    foreach(K 3 5 8)
        add_executable(${ARM}_k${K} "${TEST_SOURCE}")
        target_compile_definitions(${ARM}_k${K} PRIVATE WIREHAIR_STATIC=1 WIREHAIR_V2_SMALL_TEST_K=${K})
        target_link_libraries(${ARM}_k${K} PRIVATE ${ARM}_static)
        add_test(NAME ${ARM}_k${K} COMMAND ${ARM}_k${K})
    endforeach()
    add_executable(${ARM}_small "${SMALL_TEST_SOURCE}")
    target_compile_definitions(${ARM}_small PRIVATE WIREHAIR_STATIC=1)
    target_link_libraries(${ARM}_small PRIVATE ${ARM}_static)
    add_test(NAME ${ARM}_small COMMAND ${ARM}_small)
    add_executable(${ARM}_small_c "${ROOT}/test/SmallCConsumer.c")
    target_link_libraries(${ARM}_small_c PRIVATE ${ARM})
    add_test(NAME ${ARM}_small_c COMMAND ${ARM}_small_c)
    foreach(CASE profile borrowed compat storage)
        if(CASE STREQUAL "profile")
            set(TEST_SOURCE "${ROOT}/codec/V2ProfileTest.cpp")
        elseif(CASE STREQUAL "borrowed")
            set(TEST_SOURCE "${ROOT}/codec/V2BorrowedSourceTest.cpp")
        elseif(CASE STREQUAL "compat")
            set(TEST_SOURCE "${ROOT}/test/V2CertifiedCompatibility.cpp")
        else()
            set(TEST_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/StorageTest.cpp")
        endif()
        add_executable(${ARM}_${CASE} "${TEST_SOURCE}")
        target_compile_definitions(${ARM}_${CASE} PRIVATE WIREHAIR_STATIC=1)
        target_link_libraries(${ARM}_${CASE} PRIVATE ${ARM}_static)
        if(ARM STREQUAL "candidate" AND CASE STREQUAL "storage")
            target_include_directories(${ARM}_${CASE} BEFORE PRIVATE "${BINARY}/candidate")
            target_compile_definitions(${ARM}_${CASE} PRIVATE STORAGE_CANDIDATE=1)
        endif()
        if(NOT CASE STREQUAL "compat")
            add_test(NAME ${ARM}_${CASE} COMMAND ${ARM}_${CASE})
        endif()
    endforeach()
    add_executable(${ARM}_fault "${ROOT}/test/V2BorrowedFacadeFaultTest.cpp"
        "${PROFILE_SOURCE}" $<TARGET_OBJECTS:common>
        "${ROOT}/codec/WirehairSmall.cpp" "${ROOT}/codec/WirehairSmallK5.cpp" "${ROOT}/codec/WirehairSmallK8.cpp")
    # All small TUs in one executable must see the same private core definition.
    if(ARM STREQUAL "candidate")
        set_property(TARGET ${ARM}_fault PROPERTY SOURCES
            "${ROOT}/test/V2BorrowedFacadeFaultTest.cpp" "${PROFILE_SOURCE}" $<TARGET_OBJECTS:common>
            "${BINARY}/candidate/WirehairSmall.cpp" "${BINARY}/candidate/WirehairSmallK5.cpp"
            "${BINARY}/candidate/WirehairSmallK8.cpp")
    endif()
    target_compile_definitions(${ARM}_fault PRIVATE WIREHAIR_STATIC=1 WIREHAIR_TESTING=1)
    target_link_libraries(${ARM}_fault PRIVATE Threads::Threads)
    add_test(NAME ${ARM}_fault COMMAND ${ARM}_fault)
endforeach()
add_executable(parity "${ROOT}/bench/Wh2SmallPayload2/Screen.cpp")
target_link_libraries(parity PRIVATE ${CMAKE_DL_LIBS})
add_test(NAME public_parity COMMAND parity $<TARGET_FILE:baseline> $<TARGET_FILE:candidate> neutral)
add_test(NAME public_parity_reverse COMMAND parity $<TARGET_FILE:baseline> $<TARGET_FILE:candidate> neutral-reverse)
add_test(NAME certified_parity COMMAND "${CMAKE_COMMAND}"
    "-DBASELINE=$<TARGET_FILE:baseline_compat>" "-DCANDIDATE=$<TARGET_FILE:candidate_compat>"
    "-DOUTPUT_DIR=${BINARY}" -P "${ROOT}/bench/Wh2ProfileClassification/CheckParity.cmake")
get_property(TESTS DIRECTORY PROPERTY TESTS)
set_tests_properties(${TESTS} PROPERTIES TIMEOUT 120)
if(SANITIZE)
    set_tests_properties(${TESTS} PROPERTIES ENVIRONMENT
        "ASAN_OPTIONS=detect_leaks=1:detect_stack_use_after_return=1;UBSAN_OPTIONS=halt_on_error=1")
endif()
