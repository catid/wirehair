cmake_minimum_required(VERSION 3.20)
project(Wh2CertifiedFallthrough LANGUAGES CXX)

# Compile-only by default; optional bounded correctness checks after assembly
# review. Neither production nor a preserved object is modified. No timing.
option(WH2_FALLTHROUGH_NEUTRAL "Build existing correctness tests, never timing" OFF)
get_filename_component(_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
get_filename_component(_binary "${CMAKE_CURRENT_BINARY_DIR}" REALPATH)
string(FIND "${_binary}/" "${_root}/" _inside)
if(_inside EQUAL 0 OR CMAKE_CROSSCOMPILING OR
   NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR
   NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR
   NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "13.3.0")
    message(FATAL_ERROR "Use an external native GNU13.3/Linux build")
endif()

set(_source "${_root}/codec/WirehairV2Profile.cpp")
file(SHA256 "${_source}" _sha)
if(NOT _sha STREQUAL "3e3fd6e0dbf1f0d5d12cc2409f6b0ea71b2853d210457f76a1e3f21fc09d68b6")
    message(FATAL_ERROR "Frozen profile source changed")
endif()
file(READ "${_source}" _candidate)
function(replace_exact _old _new _count)
    string(REPLACE "${_old}" "" _removed "${_candidate}")
    string(LENGTH "${_candidate}" _before)
    string(LENGTH "${_removed}" _after)
    string(LENGTH "${_old}" _length)
    math(EXPR _expected "${_count} * ${_length}")
    math(EXPR _actual "${_before} - ${_after}")
    if(NOT _actual EQUAL _expected)
        message(FATAL_ERROR "Missing or ambiguous dispatch anchor: ${_old}")
    endif()
    string(REPLACE "${_old}" "${_new}" _result "${_candidate}")
    set(_candidate "${_result}" PARENT_SCOPE)
endfunction()

# Prefer fall-through for the general certified profile. This is a code-layout
# hypothesis only; it must not inherit historical timing or recovery claims.
replace_exact("if (codec && codec->SmallK)"
              "if (codec && CAT_UNLIKELY(codec->SmallK))" 1)
replace_exact("if (impl->SmallK)" "if (CAT_UNLIKELY(impl->SmallK))" 2)
replace_exact("const WirehairV2Result result = impl->SmallK ?"
              "const WirehairV2Result result = CAT_UNLIKELY(impl->SmallK) ?" 1)
set(_generated "${_binary}/WirehairV2ProfileFallthrough.cpp")
file(WRITE "${_generated}" "${_candidate}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_source}")

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(fallthrough_baseline OBJECT "${_source}")
add_library(fallthrough_candidate OBJECT "${_generated}")
foreach(_target IN ITEMS fallthrough_baseline fallthrough_candidate)
    set_target_properties(${_target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_definitions(${_target} PRIVATE WIREHAIR_BUILDING=1 NDEBUG)
    target_compile_options(${_target} PRIVATE -O3 -Wall -Wextra -Wpedantic -Werror)
    # The added search directories resolve the generated TU's relative includes.
    # Exact baseline object reproduction below tests their neutrality.
    target_include_directories(${_target} PRIVATE "${_root}/include" "${_root}/codec" "${_root}")
endforeach()

    COMMAND "${CMAKE_COMMAND}" -E compare_files
        "$<TARGET_OBJECTS:fallthrough_baseline>" "${_original}"
    DEPENDS fallthrough_baseline
