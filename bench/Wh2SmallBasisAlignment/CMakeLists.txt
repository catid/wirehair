cmake_minimum_required(VERSION 3.20)
project(Wh2SmallBasisAlignment LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
get_filename_component(BINARY "${CMAKE_CURRENT_BINARY_DIR}" REALPATH)
string(FIND "${BINARY}/" "${ROOT}/" INSIDE)
if(INSIDE EQUAL 0 OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Use a fresh external Linux build")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SANITIZE "Instrument the observer (use with sanitizer DSOs)" OFF)
add_compile_options(-O3 -Wall -Wextra -Wpedantic -Werror -fno-lto)
add_link_options(-fno-lto)
if(SANITIZE)
    add_compile_definitions(WH2_BASIS_ASAN=1)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
add_executable(trace Trace.cpp)
target_include_directories(trace PRIVATE "${ROOT}/include")
target_link_libraries(trace PRIVATE ${CMAKE_DL_LIBS})
# Export the normal new/delete replacements for the dynamically loaded DSOs.
set_target_properties(trace PROPERTIES ENABLE_EXPORTS ON)
add_executable(controlled Controlled.cpp)
target_include_directories(controlled PRIVATE "${ROOT}/include")
target_link_libraries(controlled PRIVATE ${CMAKE_DL_LIBS})
set_target_properties(controlled PROPERTIES ENABLE_EXPORTS ON)
