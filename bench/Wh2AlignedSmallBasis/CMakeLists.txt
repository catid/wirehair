cmake_minimum_required(VERSION 3.20)
project(Wh2AlignedSmallBasis LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
get_filename_component(BINARY "${CMAKE_CURRENT_BINARY_DIR}" REALPATH)
string(FIND "${BINARY}/" "${ROOT}/" INSIDE)
if(INSIDE EQUAL 0 OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Use a fresh external Linux build")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PORTABLE "Force portable arithmetic, not real-host speed" OFF)
option(SANITIZE "Instrument all codec and test objects" OFF)
add_compile_options(-O3 -Wall -Wextra -Wpedantic -Werror -fno-lto)
add_link_options(-fno-lto)
if(PORTABLE)
    add_compile_definitions(ANDROID=1)
endif()
if(SANITIZE)
    add_compile_definitions(WH2_BASIS_ASAN=1)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
include_directories("${ROOT}/include" "${ROOT}/codec" "${ROOT}")
find_package(Threads REQUIRED)
set(PROFILE "${ROOT}/codec/WirehairV2Profile.cpp")
file(SHA256 "${PROFILE}" PROFILE_SHA)
if(NOT PROFILE_SHA STREQUAL "975da8d892363d05de3ad4535ec79b4a386bf3f6fab377b114af393f82d96d15")
    message(FATAL_ERROR "Re-audit alignment overlay after profile source changes")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PROFILE}" "${CMAKE_CURRENT_SOURCE_DIR}/Basis.inc")
file(READ "${PROFILE}" SOURCE)
set(OLD [=[    codec->SmallBasis.reset(new (std::nothrow) uint8_t[(size_t)profile.message_bytes]);
    if (!codec->SmallBasis) {
        return WirehairV2_OOM;
    }
    std::memcpy(codec->SmallBasis.get(), message, (size_t)profile.message_bytes);
    const WirehairV2Result result = MapSmallResult(
        wirehair_small_core::Encoder<K>::Create(lookup, codec->SmallBasis.get(),
            profile.message_bytes, profile.block_bytes, codec->SmallEncoder));]=])
string(FIND "${SOURCE}" "${OLD}" FOUND)
if(FOUND LESS 0)
    message(FATAL_ERROR "Alignment insertion point missing")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Basis.inc" REPLACEMENT)
string(REPLACE "${OLD}" "${REPLACEMENT}" SOURCE "${SOURCE}")
file(MAKE_DIRECTORY "${BINARY}/candidate")
file(WRITE "${BINARY}/candidate/WirehairV2Profile.cpp" "${SOURCE}")
set(COMMON wirehair.cpp gf256.cpp WirehairCodec.cpp WirehairTools.cpp
    codec/WirehairK6.cpp codec/WirehairK6Core.cpp codec/WirehairV2Codec.cpp
    codec/WirehairV2Peel.cpp codec/WirehairV2Plan.cpp codec/WirehairV2Policy.cpp
    codec/WirehairV2Precode.cpp codec/WirehairV2PrecodeDecode.cpp
    codec/WirehairV2PrecodeEncode.cpp codec/WirehairV2Seeds.cpp codec/WirehairV2Solve.cpp
    codec/WirehairSmall.cpp codec/WirehairSmallK5.cpp codec/WirehairSmallK8.cpp)
list(TRANSFORM COMMON PREPEND "${ROOT}/")
add_library(common OBJECT ${COMMON})
add_library(baseline_objects OBJECT "${PROFILE}")
add_library(candidate_objects OBJECT "${BINARY}/candidate/WirehairV2Profile.cpp")
foreach(TARGET common baseline_objects candidate_objects)
    set_target_properties(${TARGET} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_definitions(${TARGET} PRIVATE WIREHAIR_BUILDING=1)
endforeach()
enable_testing()
foreach(ARM baseline candidate)
    add_library(${ARM}_static STATIC $<TARGET_OBJECTS:common> $<TARGET_OBJECTS:${ARM}_objects>)
    target_link_libraries(${ARM}_static PUBLIC Threads::Threads)
    add_library(${ARM} SHARED $<TARGET_OBJECTS:common> $<TARGET_OBJECTS:${ARM}_objects>)
    target_link_libraries(${ARM} PRIVATE Threads::Threads)
    target_link_options(${ARM} PRIVATE "-Wl,--version-script=${ROOT}/abi/wirehair.map")
    foreach(K 3 5 8)
        add_executable(${ARM}_k${K} "${ROOT}/test/V2SmallCodecTest.cpp")
        target_compile_definitions(${ARM}_k${K} PRIVATE WIREHAIR_STATIC=1 WIREHAIR_V2_SMALL_TEST_K=${K})
        target_link_libraries(${ARM}_k${K} PRIVATE ${ARM}_static)
        add_test(NAME ${ARM}_k${K} COMMAND ${ARM}_k${K})
        set_tests_properties(${ARM}_k${K} PROPERTIES TIMEOUT 120)
    endforeach()
    add_executable(${ARM}_allocation AllocationTest.cpp)
    target_compile_definitions(${ARM}_allocation PRIVATE WIREHAIR_STATIC=1)
    if(ARM STREQUAL "candidate")
        target_compile_definitions(${ARM}_allocation PRIVATE ALIGN_CANDIDATE=1)
    endif()
    target_link_libraries(${ARM}_allocation PRIVATE ${ARM}_static)
    add_test(NAME ${ARM}_allocation COMMAND ${ARM}_allocation)
    set_tests_properties(${ARM}_allocation PROPERTIES TIMEOUT 120)
endforeach()
# Reuse only its neutral fixture/parity path. No timing controller is provided.
add_executable(parity "${ROOT}/bench/Wh2SmallPayload2/Screen.cpp")
target_link_libraries(parity PRIVATE ${CMAKE_DL_LIBS})
add_test(NAME public_parity COMMAND parity $<TARGET_FILE:baseline> $<TARGET_FILE:candidate> neutral)
add_test(NAME public_parity_reverse COMMAND parity $<TARGET_FILE:baseline> $<TARGET_FILE:candidate> neutral-reverse)
set_tests_properties(public_parity public_parity_reverse PROPERTIES TIMEOUT 120)
