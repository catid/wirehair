cmake_minimum_required(VERSION 3.20)
project(Wh2DecoderAllocationTrace LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
get_filename_component(BINARY "${CMAKE_CURRENT_BINARY_DIR}" REALPATH)
string(FIND "${BINARY}/" "${ROOT}/" INSIDE)
if(INSIDE EQUAL 0 OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use a fresh external 64-bit Linux observer build")
endif()
set(NATURAL_SOURCE "" CACHE FILEPATH "Retained qualified Natural.cpp, never rebuilt here")
file(SHA256 "${NATURAL_SOURCE}" SOURCE_HASH)
if(NOT SOURCE_HASH STREQUAL "d97b7a4d289764c8729e752dea534a97f1fa6ffdf51f4f5f9965e8a976e1fc6c")
    message(FATAL_ERROR "Expected the exact retained aligned-basis natural worker")
endif()
file(READ "${NATURAL_SOURCE}" SOURCE)
string(FIND "${SOURCE}" "uint64_t Now()" CUT)
if(CUT LESS 0)
    message(FATAL_ERROR "Missing non-timing prefix boundary")
endif()
string(SUBSTRING "${SOURCE}" 0 ${CUT} PREFIX)
function(replace_one OLD NEW)
    string(FIND "${PREFIX}" "${OLD}" FIRST)
    if(FIRST LESS 0)
        message(FATAL_ERROR "Missing instrumentation point: ${OLD}")
    endif()
    string(LENGTH "${OLD}" LENGTH)
    math(EXPR NEXT "${FIRST}+${LENGTH}")
    string(SUBSTRING "${PREFIX}" ${NEXT} -1 TAIL)
    string(FIND "${TAIL}" "${OLD}" AGAIN)
    if(NOT AGAIN LESS 0)
        message(FATAL_ERROR "Ambiguous instrumentation point: ${OLD}")
    endif()
    string(REPLACE "${OLD}" "${NEW}" VALUE "${PREFIX}")
    set(PREFIX "${VALUE}" PARENT_SCOPE)
endfunction()
replace_one("WirehairV2Codec c = nullptr; int r = lib->decoder(p.data(), 32, &c);"
    "WirehairV2Codec c = nullptr; allocation_trace::Call(0,nullptr,p.data(),&c,32); int r = lib->decoder(p.data(), 32, &c);")
replace_one("int(lib->feed(static_cast<WirehairV2Codec>(h), id, in, n));"
    "(allocation_trace::Call(1,h,in,nullptr,n,id), int(lib->feed(static_cast<WirehairV2Codec>(h), id, in, n))); ")
replace_one("int r = lib->recover(static_cast<WirehairV2Codec>(h), out, bytes, &n);"
    "allocation_trace::Call(2,h,out,&n,bytes); int r = lib->recover(static_cast<WirehairV2Codec>(h), out, bytes, &n);")
replace_one("else lib->free(static_cast<WirehairV2Codec>(h));"
    "else { allocation_trace::Call(3,h,nullptr,nullptr,0); lib->free(static_cast<WirehairV2Codec>(h)); }")
file(READ Main.inc MAIN)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${NATURAL_SOURCE}" "${CMAKE_CURRENT_SOURCE_DIR}/Main.inc")
file(WRITE "${BINARY}/Trace.cpp" "#include \"Allocation.h\"\n${PREFIX}\n}\n${MAIN}")
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(SANITIZE "Instrument observer; pass matching existing sanitizer DSOs" OFF)
add_executable(trace "${BINARY}/Trace.cpp")
target_include_directories(trace PRIVATE "${ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_options(trace PRIVATE -O3 -Wall -Wextra -Wpedantic -Werror -fno-lto)
target_link_options(trace PRIVATE -fno-lto)
set_target_properties(trace PROPERTIES ENABLE_EXPORTS ON)
target_link_libraries(trace PRIVATE ${CMAKE_DL_LIBS})
if(SANITIZE)
    target_compile_options(trace PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
    target_link_options(trace PRIVATE -fsanitize=address,undefined)
endif()
