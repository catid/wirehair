cmake_minimum_required(VERSION 3.20)
project(Wh2K8OrdinaryShared LANGUAGES C CXX)
get_filename_component(_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
get_filename_component(_binary "${CMAKE_CURRENT_BINARY_DIR}" REALPATH)
string(FIND "${_binary}/" "${_root}/" _inside)
if(_inside EQUAL 0 OR CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux"
   OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "13.3.0")
    message(FATAL_ERROR "Use a fresh external native GNU13.3/Linux test build")
endif()
set(WH2_K8_SHARED_PREPARED "" CACHE PATH "Fresh exact native DSO preparation directory")
if(NOT EXISTS "${WH2_K8_SHARED_PREPARED}/PREPARED.json")
    message(FATAL_ERROR "Run Prepare.py into a fresh external directory first")
endif()
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/Prepare.py"
    "${WH2_K8_SHARED_PREPARED}" --verify RESULT_VARIABLE _verified)
if(NOT _verified EQUAL 0)
    message(FATAL_ERROR "Native shared preparation verification failed")
endif()
file(READ "${WH2_K8_SHARED_PREPARED}/PREPARED.json" _proof)
foreach(_index RANGE 0 1)
    string(JSON _arm GET "${_proof}" links ${_index} arm)
    string(JSON _path GET "${_proof}" links ${_index} dso path)
    string(JSON _sha GET "${_proof}" links ${_index} dso sha256)
    file(SHA256 "${_path}" _actual)
    if(NOT _actual STREQUAL _sha)
        message(FATAL_ERROR "Prepared ${_arm} shared library changed")
    endif()
    add_library(${_arm} SHARED IMPORTED)
    set_target_properties(${_arm} PROPERTIES IMPORTED_LOCATION "${_path}" IMPORTED_SONAME libwirehair.so.2)
endforeach()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

enable_testing()

function(shared_test name arm source)
    add_executable(${name} "${source}" "${CMAKE_CURRENT_SOURCE_DIR}/ProviderCheck.cpp")
    target_include_directories(${name} PRIVATE "${_root}/include" "${_root}/codec" "${_root}/test"
        "${_root}/bench/Wh2K8OrdinarySelector")
    target_compile_definitions(${name} PRIVATE WIREHAIR_DLL=1
        "WH2_SHARED_EXPECTED_PROVIDER=\"$<TARGET_FILE:${arm}>\"")
    target_compile_options(${name} PRIVATE -O3 -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(${name} PROPERTIES LINKER_LANGUAGE CXX)
    target_link_libraries(${name} PRIVATE ${arm} ${CMAKE_DL_LIBS})
    target_link_options(${name} PRIVATE "-Wl,-Map,${_binary}/${name}.map")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 120)
endfunction()

foreach(_arm IN ITEMS baseline candidate)
    foreach(_k IN ITEMS 3 5 8)
        if(_arm STREQUAL "candidate")

        else()
            set(_source "${_root}/test/V2SmallCodecTest.cpp")

        shared_test(${_arm}_k${_k} ${_arm} "${_source}")
        target_compile_definitions(${_arm}_k${_k} PRIVATE WIREHAIR_V2_SMALL_TEST_K=${_k})
    endforeach()
    shared_test(${_arm}_borrowed ${_arm} "${_root}/codec/V2BorrowedSourceTest.cpp")
    shared_test(${_arm}_metadata ${_arm} "${_root}/bench/Wh2ProfileClassification/ValidationTest.cpp")

    shared_test(${_arm}_legacy_cpp ${_arm} "${_root}/test/LegacyCppConsumer.cpp")
    shared_test(${_arm}_small_c ${_arm} "${_root}/test/SmallCConsumer.c")
    shared_test(${_arm}_k6_c ${_arm} "${_root}/test/K6CConsumer.c")
    shared_test(${_arm}_k3_c ${_arm} "${_root}/test/V2SmallCConsumer.c")

    shared_test(${_arm}_k8_explicit_c ${_arm} "${_root}/test/V2SmallK8CConsumer.c")
    # Byte emitters are run by the parity checker, not as noisy standalone tests.
    add_executable(${_arm}_certified "${_root}/test/V2CertifiedCompatibility.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/ProviderCheck.cpp")
    target_include_directories(${_arm}_certified PRIVATE "${_root}/include")
    target_compile_options(${_arm}_certified PRIVATE -O3 -Wall -Wextra -Wpedantic -Werror)
    target_compile_definitions(${_arm}_certified PRIVATE WIREHAIR_DLL=1
        "WH2_SHARED_EXPECTED_PROVIDER=\"$<TARGET_FILE:${_arm}>\"")
    target_link_libraries(${_arm}_certified PRIVATE ${_arm} ${CMAKE_DL_LIBS})
    target_link_options(${_arm}_certified PRIVATE "-Wl,-Map,${_binary}/${_arm}_certified.map")
    add_test(NAME ${_arm}_exports COMMAND "${CMAKE_COMMAND}"
        "-DLIBRARY=$<TARGET_FILE:${_arm}>" "-DMANIFEST=${_root}/abi/wirehair.map" "-DNM=${CMAKE_NM}"
        -P "${_root}/test/cmake/CheckElfExports.cmake")
endforeach()
