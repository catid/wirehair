cmake_minimum_required(VERSION 3.20)
project(Wh2K8RelocatedShared LANGUAGES C CXX)
get_filename_component(_root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." REALPATH)
get_filename_component(_binary "${CMAKE_CURRENT_BINARY_DIR}" REALPATH)
string(FIND "${_binary}/" "${_root}/" _inside)
if(_inside EQUAL 0 OR CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux"
   OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "13.3.0")
    message(FATAL_ERROR "Use a fresh external native GNU13.3/Linux package build")
endif()
if(NOT WH2_PACKAGE_ARM MATCHES "^(baseline|candidate)$" OR NOT IS_DIRECTORY "${WH2_RELOCATED_PREFIX}")
    message(FATAL_ERROR "An explicit arm and relocated prefix are required")
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
set(wirehair_DIR "${WH2_RELOCATED_PREFIX}/lib64/wirehair-e2e/cmake/wirehair" CACHE PATH "Exact relocated package" FORCE)
find_package(wirehair 2.0.0 EXACT REQUIRED CONFIG PATHS "${wirehair_DIR}" NO_DEFAULT_PATH)
get_target_property(_type wirehair::wirehair TYPE)
get_target_property(_definitions wirehair::wirehair INTERFACE_COMPILE_DEFINITIONS)
get_target_property(_features wirehair::wirehair INTERFACE_COMPILE_FEATURES)
get_target_property(_includes wirehair::wirehair INTERFACE_INCLUDE_DIRECTORIES)
get_target_property(_location wirehair::wirehair IMPORTED_LOCATION_RELEASE)
set(_expected "${WH2_RELOCATED_PREFIX}/lib64/wirehair-e2e/libwirehair.so.2.0.0")
if(NOT _type STREQUAL "SHARED_LIBRARY" OR NOT _definitions STREQUAL "WIREHAIR_DLL=1"
   OR NOT _features STREQUAL "cxx_std_11" OR NOT _includes STREQUAL "${WH2_RELOCATED_PREFIX}/include"
   OR NOT _location STREQUAL _expected)
    message(FATAL_ERROR "Unexpected installed target type/interface/location")
endif()
file(WRITE "${_binary}/imported-target.txt" "${_type}\n${_definitions}\n${_features}\n${_includes}\n${_location}\n")
enable_testing()
set(EXPECTED_WIREHAIR_VERSION 2.0.0)
set(WIREHAIR_STRICT_WARNINGS ON CACHE BOOL "Strict installed consumers" FORCE)
add_subdirectory("${_root}/test/package" upstream)

add_executable(package_ordinary_cpp "${CMAKE_CURRENT_SOURCE_DIR}/OrdinaryConsumer.cpp")
target_link_libraries(package_ordinary_cpp PRIVATE wirehair::wirehair)
set_target_properties(package_ordinary_cpp PROPERTIES CXX_STANDARD 98 CXX_EXTENSIONS OFF)
add_test(NAME package_ordinary_cpp COMMAND package_ordinary_cpp)
if(WH2_PACKAGE_ARM STREQUAL "candidate")
    target_compile_definitions(package_ordinary_cpp PRIVATE WH2_EXPECT_SMALL_K8=1)
    set(_c_source "${_root}/test/V2SmallK8CConsumer.c")
    file(SHA256 "${_c_source}" _c_sha)
    if(NOT _c_sha STREQUAL "93b3475fb609db457c05bf206d07eb224afd4532774607f9b47c31948d788163")
        message(FATAL_ERROR "Frozen literal K8 C consumer changed")
    endif()
    file(READ "${_c_source}" _c)
    string(REPLACE "wirehair_v2_encoder_create_profile_id_with_options(WIREHAIR_V2_PROFILE_SMALL_K8_2026_09,"
        "wirehair_v2_encoder_create_with_options(" _c "${_c}")
    file(WRITE "${_binary}/ordinary.c" "${_c}")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_c_source}")
    add_executable(package_ordinary_c "${_binary}/ordinary.c")
    target_link_libraries(package_ordinary_c PRIVATE wirehair::wirehair)
    add_test(NAME package_ordinary_c COMMAND package_ordinary_c)
else()
    target_compile_definitions(package_ordinary_cpp PRIVATE WH2_EXPECT_SMALL_K8=0)
endif()

set(_consumers package_c_consumer package_cpp_consumer package_k6_consumer package_small_consumer
    package_v2_small_consumer package_v2_k5_consumer package_v2_k8_consumer package_plugin package_ordinary_cpp)
if(WH2_PACKAGE_ARM STREQUAL "candidate")
    list(APPEND _consumers package_ordinary_c)
endif()
foreach(_consumer IN LISTS _consumers)
    target_sources(${_consumer} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../ProviderCheck.cpp")
    target_compile_definitions(${_consumer} PRIVATE "WH2_SHARED_EXPECTED_PROVIDER=\"${_expected}\"")
    target_compile_options(${_consumer} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_link_libraries(${_consumer} PRIVATE ${CMAKE_DL_LIBS})
    set_target_properties(${_consumer} PROPERTIES LINKER_LANGUAGE CXX)
    target_link_options(${_consumer} PRIVATE "-Wl,-Map,${_binary}/${_consumer}.map")
endforeach()
target_link_options(package_plugin_host PRIVATE "-Wl,-Map,${_binary}/package_plugin_host.map")
get_property(_local_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${_local_tests} PROPERTIES TIMEOUT 30)
