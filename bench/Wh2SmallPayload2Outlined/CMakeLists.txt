cmake_minimum_required(VERSION 3.20)
project(Wh2SmallPayload2Outlined LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
get_filename_component(BINARY "${CMAKE_CURRENT_BINARY_DIR}" REALPATH)
string(FIND "${BINARY}/" "${ROOT}/" INSIDE)
if(INSIDE EQUAL 0 OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Use a fresh external Linux build")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PORTABLE "Force portable arithmetic, not real-host speed" OFF)
option(SANITIZE "Instrument all codec and observer objects" OFF)
option(COUNT "Check the existing WH_COUNT fallback" OFF)
add_compile_options(-O3 -Wall -Wextra -Wpedantic -Werror -fno-lto)
add_link_options(-fno-lto)
if(PORTABLE)
    add_compile_definitions(ANDROID=1)
endif()
if(COUNT)
    add_compile_definitions(WH_COUNT=1)
endif()
if(SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
include_directories("${ROOT}/include" "${ROOT}/codec" "${ROOT}")
find_package(Threads REQUIRED)

file(SHA256 "${ROOT}/codec/WirehairSmallCore.h" CORE_SHA)
if(NOT CORE_SHA STREQUAL "5b0acdd096d24b76351bacd1718c44ca5b37d4df587fe7334822a9f61f1e0b8c")
    message(FATAL_ERROR "Re-audit overlay after shared-core changes")
endif()
file(READ "${ROOT}/codec/WirehairSmallCore.h" CORE)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ROOT}/codec/WirehairSmallCore.h")
set(OLD_CALL "wirehair_k6_payload::Payload(output, sources_, coefficients, K, static_cast<int>(block_bytes_));")
set(NEW_CALL "wh2_small_payload2_outlined::Payload<K>(output, sources_, coefficients, static_cast<int>(block_bytes_));")
string(REPLACE "${OLD_CALL}" "${NEW_CALL}" CORE "${CORE}")
string(REPLACE "#include \"WirehairK6Payload.h\""
    "#include \"${CMAKE_CURRENT_SOURCE_DIR}/Payload.h\"" CORE "${CORE}")
file(MAKE_DIRECTORY "${BINARY}/candidate")
file(WRITE "${BINARY}/candidate/WirehairSmallCore.h" "${CORE}")
foreach(NAME WirehairSmallLookup.h WirehairSmallFacade.h)
    configure_file("${ROOT}/codec/${NAME}" "${BINARY}/candidate/${NAME}" COPYONLY)
endforeach()
set(SMALL WirehairV2Profile.cpp WirehairSmall.cpp WirehairSmallK5.cpp WirehairSmallK8.cpp)
foreach(NAME IN LISTS SMALL)
    configure_file("${ROOT}/codec/${NAME}" "${BINARY}/candidate/${NAME}" COPYONLY)
    list(APPEND BASE_SOURCES "${ROOT}/codec/${NAME}")
    list(APPEND CANDIDATE_SOURCES "${BINARY}/candidate/${NAME}")
endforeach()
set(COMMON wirehair.cpp gf256.cpp WirehairCodec.cpp WirehairTools.cpp
    codec/WirehairK6.cpp codec/WirehairK6Core.cpp codec/WirehairV2Codec.cpp
    codec/WirehairV2Peel.cpp codec/WirehairV2Plan.cpp codec/WirehairV2Policy.cpp
    codec/WirehairV2Precode.cpp codec/WirehairV2PrecodeDecode.cpp
    codec/WirehairV2PrecodeEncode.cpp codec/WirehairV2Seeds.cpp codec/WirehairV2Solve.cpp)
list(TRANSFORM COMMON PREPEND "${ROOT}/")
add_library(common OBJECT ${COMMON})
add_library(baseline_objects OBJECT ${BASE_SOURCES})
add_library(candidate_objects OBJECT ${CANDIDATE_SOURCES} Payload.cpp)
foreach(TARGET common baseline_objects candidate_objects)
    set_target_properties(${TARGET} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_definitions(${TARGET} PRIVATE WIREHAIR_BUILDING=1)
endforeach()
enable_testing()
foreach(ARM baseline candidate)
    add_library(${ARM}_static STATIC $<TARGET_OBJECTS:common> $<TARGET_OBJECTS:${ARM}_objects>)
    target_link_libraries(${ARM}_static PUBLIC Threads::Threads)
    add_library(${ARM} SHARED $<TARGET_OBJECTS:common> $<TARGET_OBJECTS:${ARM}_objects>)
    target_link_libraries(${ARM} PRIVATE Threads::Threads)
    target_link_options(${ARM} PRIVATE "-Wl,--version-script=${ROOT}/abi/wirehair.map")
    foreach(K 3 5 8)
        add_executable(${ARM}_k${K} "${ROOT}/test/V2SmallCodecTest.cpp")
        target_compile_definitions(${ARM}_k${K} PRIVATE WIREHAIR_STATIC=1 WIREHAIR_V2_SMALL_TEST_K=${K})
        target_link_libraries(${ARM}_k${K} PRIVATE ${ARM}_static)
        add_test(NAME ${ARM}_k${K} COMMAND ${ARM}_k${K})
        set_tests_properties(${ARM}_k${K} PROPERTIES TIMEOUT 120)
    endforeach()
endforeach()
# Reuse the independent exhaustive/fallback/redzone oracle, not its candidate.
set(ORACLE "${ROOT}/bench/Wh2SmallPayload2/PayloadTest.cpp")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ORACLE}")
file(READ "${ORACLE}" TEST)
string(REPLACE "#include \"Payload.h\"" "#include \"${CMAKE_CURRENT_SOURCE_DIR}/Payload.h\"" TEST "${TEST}")
string(REPLACE "wh2_small_payload2::" "wh2_small_payload2_outlined::" TEST "${TEST}")
file(WRITE "${BINARY}/PayloadTest.cpp" "${TEST}")
add_executable(payload_test "${BINARY}/PayloadTest.cpp")
target_link_libraries(payload_test PRIVATE candidate_static)
add_test(NAME payload_test COMMAND payload_test)
set_tests_properties(payload_test PROPERTIES TIMEOUT 120)
add_executable(dispatch_test DispatchTest.cpp)
target_link_libraries(dispatch_test PRIVATE candidate_static)
add_test(NAME dispatch_test COMMAND dispatch_test)
set_tests_properties(dispatch_test PROPERTIES TIMEOUT 120)
# Reuse the observer with an explicitly different address-matching contract.
# Never edit the spent experiment's worker. The fixed source digest makes the
# transformations fail closed if its fixtures/loops are changed in the future.
set(OBSERVER "${ROOT}/bench/Wh2SmallPayload2/Screen.cpp")
file(SHA256 "${OBSERVER}" OBSERVER_SHA)
if(NOT OBSERVER_SHA STREQUAL "59a3e1d69fc2230bf1119ec0b8c209ef7f669ad1a613c96ee40a12f23b433494")
    message(FATAL_ERROR "Re-audit matched workspace overlay after observer changes")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${OBSERVER}")
file(READ "${OBSERVER}" WORKER)
string(REPLACE "const auto& fs = cells[cell/4];" [=[const auto& fs = cells[cell/4];
                // All arms borrow the same immutable source address. Staging
                // lives until after all prebuilt handles have been destroyed.
                Fixture matched = fs[0];
                for (unsigned a = 0; a < 3; ++a)
                    Check(matched.source == fs[a].source, "matched source contents");]=] WORKER "${WORKER}")
string(REPLACE "fs[a].shape,fs[a].source.data(),p" "fs[a].shape,matched.source.data(),p" WORKER "${WORKER}")
string(REPLACE "std::vector<Byte> output[2] = {std::vector<Byte>(size+32,173),std::vector<Byte>(size+32,173)};"
    "std::vector<Byte> output(size+32,173);" WORKER "${WORKER}")
string(REPLACE "std::fill(output[side].begin()+16,output[side].end()-16,0);" [=[// Decoder payload equations/counts remain arm-specific, but
                        // the input and output addresses are shared. All copies
                        // and verification are outside WORK.
                        matched.profile = fs[arm].profile;
                        for (unsigned family = 0; family < 2; ++family) {
                            Check(matched.repairs[family].size() == fs[arm].repairs[family].size(), "staging size");
                            std::copy(fs[arm].repairs[family].begin(), fs[arm].repairs[family].end(), matched.repairs[family].begin());
                            matched.steps[family] = fs[arm].steps[family];
                        }
                        std::fill(output.begin()+16,output.end()-16,0);]=] WORKER "${WORKER}")
string(REPLACE "Work(apis[arm], fs[arm]," "Work(apis[arm], matched," WORKER "${WORKER}")
string(REPLACE "output[side]" "output" WORKER "${WORKER}")
file(WRITE "${BINARY}/candidate/Screen.cpp" "${WORKER}")
add_executable(screen "${BINARY}/candidate/Screen.cpp")
target_link_libraries(screen PRIVATE ${CMAKE_DL_LIBS})
# Only neutral parity is launched. Do not use the original spent screen runner.
add_test(NAME public_parity COMMAND screen $<TARGET_FILE:baseline> $<TARGET_FILE:candidate> neutral)
add_test(NAME public_parity_reverse COMMAND screen $<TARGET_FILE:baseline> $<TARGET_FILE:candidate> neutral-reverse)
set_tests_properties(public_parity public_parity_reverse PROPERTIES TIMEOUT 120)
add_test(NAME identical_library_rejected COMMAND screen $<TARGET_FILE:baseline> $<TARGET_FILE:baseline> neutral)
set_tests_properties(identical_library_rejected PROPERTIES TIMEOUT 120 WILL_FAIL TRUE)
