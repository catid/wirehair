cmake_minimum_required(VERSION 3.16)
project(Wh2ThueMorseTinyPayloadR0 LANGUAGES CXX)

# Standalone neutral-only build. No production/default target is changed.
option(WH2_TINY_PAYLOAD_SANITIZERS "Enable ASan/UBSan for neutral tests" OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR CMAKE_CROSSCOMPILING OR
   NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR CMAKE_CONFIGURATION_TYPES)
    message(FATAL_ERROR "This prototype uses native single-configuration Linux/GNU C++")
endif()
get_filename_component(_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
get_filename_component(_binary "${CMAKE_CURRENT_BINARY_DIR}" REALPATH)
string(FIND "${_binary}/" "${_root}/" _inside_source)
if(_inside_source EQUAL 0)
    message(FATAL_ERROR "Use a fresh build directory outside the source tree")
endif()
string(TOUPPER "${CMAKE_BUILD_TYPE}" _config)
set(_flags "${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_${_config}} ${CMAKE_EXE_LINKER_FLAGS}")
if(CMAKE_INTERPROCEDURAL_OPTIMIZATION OR CMAKE_INTERPROCEDURAL_OPTIMIZATION_${_config} OR
   _flags MATCHES "WH_COUNT|WIREHAIR_TESTING|WIREHAIR_V2_ENABLE_TEST_HOOKS|-flto|-fprofile|-march=|-mavx|-mgfni")
    message(FATAL_ERROR "Do not inject counters, hooks, LTO, PGO or ISA flags")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
find_package(Threads REQUIRED)
enable_testing()

add_library(wh2_tiny_payload_gf STATIC "${_root}/gf256.cpp")
target_link_libraries(wh2_tiny_payload_gf PUBLIC Threads::Threads)
add_library(wh2_tiny_payload_baseline OBJECT "${_root}/bench/Wh2ThueMorseNativeCodec.cpp")
add_library(wh2_tiny_payload_candidate OBJECT "${_root}/bench/Wh2ThueMorseTinyPayloadR0.cpp")
add_executable(wh2_tiny_payload_baseline_test
    "${_root}/bench/Wh2ThueMorseNativeCodecTest.cpp"
    $<TARGET_OBJECTS:wh2_tiny_payload_baseline>)
add_executable(wh2_tiny_payload_candidate_test
    "${_root}/bench/Wh2ThueMorseTinyPayloadR0CodecTest.cpp"
    $<TARGET_OBJECTS:wh2_tiny_payload_candidate>)
add_executable(wh2_tiny_payload_kernel_test "${_root}/bench/Wh2ThueMorseTinyPayloadR0Test.cpp")
set(_tests wh2_tiny_payload_baseline_test wh2_tiny_payload_candidate_test wh2_tiny_payload_kernel_test)
foreach(_target IN ITEMS wh2_tiny_payload_gf wh2_tiny_payload_baseline wh2_tiny_payload_candidate ${_tests})
    target_include_directories(${_target} PRIVATE "${_root}" "${_root}/bench")
    set_target_properties(${_target} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_options(${_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(WH2_TINY_PAYLOAD_SANITIZERS)
        target_compile_options(${_target} PRIVATE -O1 -g -fno-omit-frame-pointer
            -fsanitize=address,undefined -fno-pie)
    else()
        target_compile_options(${_target} PRIVATE -O3)
    endif()
endforeach()
foreach(_target IN LISTS _tests)
    target_link_libraries(${_target} PRIVATE wh2_tiny_payload_gf)
    if(WH2_TINY_PAYLOAD_SANITIZERS)
        target_link_options(${_target} PRIVATE -fsanitize=address,undefined -no-pie)
    endif()
    add_test(NAME ${_target} COMMAND ${_target})
    set_tests_properties(${_target} PROPERTIES TIMEOUT 60)
endforeach()
