cmake_minimum_required(VERSION 3.20)
project(Wh2SmallDormantCore LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." REALPATH)
get_filename_component(BINARY "${CMAKE_CURRENT_BINARY_DIR}" REALPATH)
string(FIND "${BINARY}/" "${ROOT}/" INSIDE)
if(INSIDE EQUAL 0 OR NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR
   NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "Use a fresh external 64-bit GNU/Linux qualification build")
endif()
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(PORTABLE "Force portable codec arithmetic" OFF)
option(SANITIZE "Instrument all codec/test objects" OFF)
add_compile_options(-O3 -Wall -Wextra -Wpedantic -Werror -fno-lto)
add_link_options(-fno-lto)
if(PORTABLE)
    add_compile_definitions(ANDROID=1)
endif()
if(SANITIZE)
    add_compile_definitions(WH2_BASIS_ASAN=1)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
include_directories("${ROOT}/include" "${ROOT}/codec" "${ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Threads REQUIRED)
set(PROFILE "${ROOT}/codec/WirehairV2Profile.cpp")
file(SHA256 "${PROFILE}" HASH)
if(NOT HASH STREQUAL "975da8d892363d05de3ad4535ec79b4a386bf3f6fab377b114af393f82d96d15")
    message(FATAL_ERROR "Re-audit dormant-core overlay after profile source changes")
endif()
file(READ "${PROFILE}" BASELINE)
set(OLD [=[    explicit PublicCodec(uint8_t small_k = 0) : SmallK(small_k) {}

    wirehair_v2::Codec Impl;]=])
string(FIND "${BASELINE}" "${OLD}" FOUND)
if(FOUND LESS 0)
    message(FATAL_ERROR "Missing exact PublicCodec declaration")
endif()
file(READ Core.inc REPLACEMENT)
string(REPLACE "${OLD}" "${REPLACEMENT}" CANDIDATE "${BASELINE}")
file(MAKE_DIRECTORY "${BINARY}/candidate")
file(WRITE "${BINARY}/candidate/WirehairV2Profile.cpp" "${CANDIDATE}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${PROFILE}" "${CMAKE_CURRENT_SOURCE_DIR}/Core.inc" "${CMAKE_CURRENT_SOURCE_DIR}/Layout.inc")
set(COMMON wirehair.cpp gf256.cpp WirehairCodec.cpp WirehairTools.cpp
    codec/WirehairK6.cpp codec/WirehairK6Core.cpp codec/WirehairV2Codec.cpp
    codec/WirehairV2Peel.cpp codec/WirehairV2Plan.cpp codec/WirehairV2Policy.cpp
    codec/WirehairV2Precode.cpp codec/WirehairV2PrecodeDecode.cpp
    codec/WirehairV2PrecodeEncode.cpp codec/WirehairV2Seeds.cpp codec/WirehairV2Solve.cpp
    codec/WirehairSmall.cpp codec/WirehairSmallK5.cpp codec/WirehairSmallK8.cpp)
list(TRANSFORM COMMON PREPEND "${ROOT}/")
add_library(common OBJECT ${COMMON})
set_target_properties(common PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(common PRIVATE WIREHAIR_BUILDING=1)
enable_testing()
foreach(ARM baseline candidate)
    if(ARM STREQUAL "baseline")
        set(SOURCE "${PROFILE}")
        set(LAYOUT_SOURCE "${BASELINE}")
    else()
        set(SOURCE "${BINARY}/candidate/WirehairV2Profile.cpp")
        set(LAYOUT_SOURCE "${CANDIDATE}")
    endif()
    add_library(${ARM}_objects OBJECT "${SOURCE}")
    set_target_properties(${ARM}_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_definitions(${ARM}_objects PRIVATE WIREHAIR_BUILDING=1)
    add_library(${ARM}_static STATIC $<TARGET_OBJECTS:common> $<TARGET_OBJECTS:${ARM}_objects>)
    target_link_libraries(${ARM}_static PUBLIC Threads::Threads)
    add_library(${ARM} SHARED $<TARGET_OBJECTS:common> $<TARGET_OBJECTS:${ARM}_objects>)
    target_link_libraries(${ARM} PRIVATE Threads::Threads)
    target_link_options(${ARM} PRIVATE "-Wl,--version-script=${ROOT}/abi/wirehair.map")
    foreach(K 3 5 8)
        add_executable(${ARM}_k${K} "${ROOT}/test/V2SmallCodecTest.cpp")
        target_compile_definitions(${ARM}_k${K} PRIVATE WIREHAIR_STATIC=1 WIREHAIR_V2_SMALL_TEST_K=${K})
        target_link_libraries(${ARM}_k${K} PRIVATE ${ARM}_static)
        add_test(NAME ${ARM}_k${K} COMMAND ${ARM}_k${K})
    endforeach()
    foreach(CASE profile borrowed allocation lifetime compat)
        if(CASE STREQUAL "profile")
            set(TEST_SOURCE "${ROOT}/codec/V2ProfileTest.cpp")
        elseif(CASE STREQUAL "borrowed")
            set(TEST_SOURCE "${ROOT}/codec/V2BorrowedSourceTest.cpp")
        elseif(CASE STREQUAL "allocation")
            # Both arms use UNCHANGED (not aligned) basis allocation semantics.
            set(TEST_SOURCE "${ROOT}/bench/Wh2AlignedSmallBasis/AllocationTest.cpp")
        elseif(CASE STREQUAL "lifetime")
            set(TEST_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/Lifetime.cpp")
        else()
            set(TEST_SOURCE "${ROOT}/test/V2CertifiedCompatibility.cpp")
        endif()
        add_executable(${ARM}_${CASE} "${TEST_SOURCE}")
        target_compile_definitions(${ARM}_${CASE} PRIVATE WIREHAIR_STATIC=1)
        target_link_libraries(${ARM}_${CASE} PRIVATE ${ARM}_static)
        if(CASE STREQUAL "lifetime")
            if(ARM STREQUAL "candidate")
                target_compile_definitions(${ARM}_${CASE} PRIVATE DORMANT_CANDIDATE=1)
            endif()
            foreach(SYMBOL _ZN11wirehair_v25CodecC1Ev _ZN11wirehair_v25CodecC2Ev
                           _ZN11wirehair_v25CodecD1Ev _ZN11wirehair_v25CodecD2Ev)
                target_link_options(${ARM}_${CASE} PRIVATE "-Wl,--wrap=${SYMBOL}")
            endforeach()
        endif()
        if(NOT CASE STREQUAL "compat")
            add_test(NAME ${ARM}_${CASE} COMMAND ${ARM}_${CASE})
        endif()
    endforeach()
    # Layout observations and forced facade-OOM hooks are test-only objects.
    # Neither is linked into the candidate DSO/archive used for future screening.
    file(WRITE "${BINARY}/${ARM}Layout.cpp" "${LAYOUT_SOURCE}\n#include \"Layout.inc\"\n")
    add_executable(${ARM}_layout "${BINARY}/${ARM}Layout.cpp" $<TARGET_OBJECTS:common>)
    target_compile_definitions(${ARM}_layout PRIVATE WIREHAIR_STATIC=1)
    target_link_libraries(${ARM}_layout PRIVATE Threads::Threads)
    add_test(NAME ${ARM}_layout COMMAND ${ARM}_layout)
    add_executable(${ARM}_fault "${ROOT}/test/V2BorrowedFacadeFaultTest.cpp" "${SOURCE}" $<TARGET_OBJECTS:common>)
    target_compile_definitions(${ARM}_fault PRIVATE WIREHAIR_STATIC=1 WIREHAIR_TESTING=1)
    target_link_libraries(${ARM}_fault PRIVATE Threads::Threads)
    add_test(NAME ${ARM}_fault COMMAND ${ARM}_fault)
endforeach()
add_executable(parity "${ROOT}/bench/Wh2SmallPayload2/Screen.cpp")
target_link_libraries(parity PRIVATE ${CMAKE_DL_LIBS})
add_test(NAME public_parity COMMAND parity $<TARGET_FILE:baseline> $<TARGET_FILE:candidate> neutral)
add_test(NAME public_parity_reverse COMMAND parity $<TARGET_FILE:baseline> $<TARGET_FILE:candidate> neutral-reverse)
add_test(NAME certified_parity COMMAND "${CMAKE_COMMAND}"
    "-DBASELINE=$<TARGET_FILE:baseline_compat>" "-DCANDIDATE=$<TARGET_FILE:candidate_compat>"
    "-DOUTPUT_DIR=${BINARY}" -P "${ROOT}/bench/Wh2ProfileClassification/CheckParity.cmake")
get_property(TESTS DIRECTORY PROPERTY TESTS)
set_tests_properties(${TESTS} PROPERTIES TIMEOUT 120)
if(SANITIZE)
    set_tests_properties(${TESTS} PROPERTIES ENVIRONMENT
        "ASAN_OPTIONS=detect_leaks=1:detect_stack_use_after_return=1;UBSAN_OPTIONS=halt_on_error=1")
endif()
